async function api(fn, ...args) {

}

       { k: 'members', i: 'people', t: 'Members' }, fund,
       { k: 'account', i: 'person-gear', t: 'Account' }]

/* =================== PROOF + ACTIONS =================== */
let blobUrl = null;
async function showProof(sheet, id, isAdmin) {
  $('proofBody').innerHTML = '<div class="spinner-border text-success my-5"></div>';
  $('proofLink').style.display = 'none';
  modal('proofModal').show();
  try {
    const p = await api('getProof', S.token, sheet, id);
    if (blobUrl) { URL.revokeObjectURL(blobUrl); blobUrl = null; }
    if (p.base64 && p.mime.indexOf('image/') === 0) {
      $('proofBody').innerHTML = `<img class="img-fluid rounded" src="data:${esc(p.mime)};base64,${p.base64}">`;
    } else if (p.base64 && p.mime === 'application/pdf') {
      const bin = atob(p.base64), arr = new Uint8Array(bin.length);
      for (let i = 0; i < bin.length; i++) arr[i] = bin.charCodeAt(i);
      blobUrl = URL.createObjectURL(new Blob([arr], { type: 'application/pdf' }));
      $('proofBody').innerHTML = `<iframe src="${blobUrl}" style="width:100%;height:70vh;border:0"></iframe>`;
    } else {
      $('proofBody').innerHTML = '<p class="text-muted my-4">Preview not available. Use “Open in Drive”.</p>';
    }
    $('proofLink').href = p.url; $('proofLink').style.display = '';
    if (isAdmin) {
      S.viewed[id] = true;
      const b = document.querySelector(`[data-approve="${CSS.escape(id)}"]`);
      if (b) { b.disabled = false; const t = b.closest('[data-card]').querySelector('small:last-child'); if (t) t.textContent = 'Proof verified — you can approve.'; }
    }
  } catch (e) { modal('proofModal').hide(); handleErr(e); }
}

  $('actTitle').textContent = title; $('actText').textContent = text;
  $('actNote').style.display = note ? '' : 'none'; $('actNote').value = '';
  $('actOk').textContent = btn; $('actOk').className = 'btn flex-fill ' + cls;
  $('actOk').onclick = async () => { modal('actModal').hide(); busy(true); try { await onOk($('actNote').value); } catch (e) { handleErr(e); } finally { busy(false); } };
  modal('actModal').show();
}
const ownSheet = () => S.data.user.sheet;
const findMember = key => S.data.admin.members.find(m => m.key === key);

$('tabContent').addEventListener('click', async e => {
  const tabLink = e.target.closest('.js-tab');
  if (tabLink) { e.preventDefault(); S.tab = tabLink.dataset.tab; render(); window.scrollTo(0, 0); return; }
  const goEl = e.target.closest('[data-go]');
  if (goEl) return openDetail(goEl.dataset.go);
  const own = e.target.closest('.js-own-proof');
  if (own) { e.preventDefault(); showProof(ownSheet(), own.dataset.id, false); return; }
  const b = e.target.closest('button'); if (!b) return;
  const { sheet, id, key } = b.dataset;
  const cl = b.classList;

  if (cl.contains('js-view')) return showProof(sheet, id, true);
  if (cl.contains('js-approve') && !b.disabled) return ask({
    title: 'Approve this entry?', text: 'Confirm that the payment proof matches the amount and date.', btn: 'Approve', cls: 'btn-success',
    onOk: async () => { await api('approveEntry', S.token, sheet, id); toast('Entry approved'); await load(true); }
  });
  if (cl.contains('js-reject')) return ask({
    title: 'Reject this entry?', text: 'The member will see it as Rejected.', note: true, btn: 'Reject', cls: 'btn-danger',
    onOk: async note => { await api('rejectEntry', S.token, sheet, id, note); toast('Entry rejected', 'warning'); await load(true); }
  });

  if (cl.contains('js-uadd')) return openUser(null);
  if (cl.contains('js-uedit')) return openUser(findMember(key));
  if (cl.contains('js-ledger')) return openLedger(sheet);
  if (cl.contains('js-ustatus')) {
    const m = findMember(key), status = b.dataset.status;
    return ask({
      title: status === 'Removed' ? 'Remove ' + m.name + '?' : 'Restore ' + m.name + '?',

      onOk: async () => { await api('setUserStatus', S.token, key, status); toast(status === 'Removed' ? 'User removed' : 'User restored'); await load(true, true); }
    });
  }
  if (cl.contains('js-chgpin')) {
    const o = $('pOld').value, n = $('pNew').value, n2 = $('pNew2').value;
    if (!/^\d{4,8}$/.test(n)) return toast('New PIN must be 4–8 digits.', 'warning');
    if (n !== n2) return toast('The new PINs do not match.', 'warning');
    busy(true);
    try { await api('changePin', S.token, o, n); toast('PIN updated'); ['pOld', 'pNew', 'pNew2'].forEach(i => $(i).value = ''); await load(true, true); }
    catch (err) { handleErr(err); } finally { busy(false); }
  }
});

function readB64(file) {
  return new Promise((res, rej) => { const r = new FileReader(); r.onload = () => res(String(r.result).split(',')[1]); r.onerror = rej; r.readAsDataURL(file); });
}
async function prepFile(f) {
  const max = (S.data.maxFileMB || 5) * 1024 * 1024;
  if (f.type === 'application/pdf') {
    if (f.size > max) throw new Error('PDF is larger than ' + S.data.maxFileMB + ' MB.');
    return { name: f.name, mime: 'application/pdf', base64: await readB64(f) };
  }
  if (f.type.indexOf('image/') !== 0) throw new Error('Only images or PDF files are allowed.');
  const url = URL.createObjectURL(f);
  try {
    const img = await new Promise((res, rej) => { const i = new Image(); i.onload = () => res(i); i.onerror = () => rej(new Error('This image format is not supported. Use JPG or PNG.')); i.src = url; });
    const scale = Math.min(1, 1600 / Math.max(img.width, img.height));
    const cv = document.createElement('canvas');
    cv.width = Math.round(img.width * scale); cv.height = Math.round(img.height * scale);
    const ctx = cv.getContext('2d'); ctx.fillStyle = '#fff'; ctx.fillRect(0, 0, cv.width, cv.height); ctx.drawImage(img, 0, 0, cv.width, cv.height);
    const b64 = cv.toDataURL('image/jpeg', 0.82).split(',')[1];
    return { name: f.name.replace(/\.\w+$/, '') + '.jpg', mime: 'image/jpeg', base64: b64 };
  } finally { URL.revokeObjectURL(url); }
}

function copyText(t) {
  const done = () => toast('কপি হয়েছে');
  if (navigator.clipboard && navigator.clipboard.writeText) navigator.clipboard.writeText(t).then(done).catch(() => fallbackCopy(t, done));
  else fallbackCopy(t, done);
}
function fallbackCopy(t, cb) {
  const ta = document.createElement('textarea'); ta.value = t; ta.style.position = 'fixed'; ta.style.opacity = '0';
  document.body.appendChild(ta); ta.focus(); ta.select();
  try { document.execCommand('copy'); cb(); } catch (e) { /* ignore */ }
  document.body.removeChild(ta);
}
